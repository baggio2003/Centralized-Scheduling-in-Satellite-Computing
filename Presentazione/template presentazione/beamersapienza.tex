\begin{frame}{Orbital Edge Computing}
\framesubtitle{From passive relays to computing in orbit}
\begin{itemize}\setlength\itemsep{4pt}
  \item \textbf{Bent-pipe model}: satellites forward raw data to ground stations $\Rightarrow$ downlink bottleneck, long delays
  \item \textbf{Orbital Edge Computing (OEC)}: onboard processors compute \emph{in space}; only results are downlinked
  \item \textbf{LEO megaconstellations} (Starlink, OneWeb, Kuiper)
  \begin{itemize}
    \item 300--2000 km altitude, $\approx$7.5 km/s $\Rightarrow$ continuously changing topology
    \item Inter-Satellite Links (ISLs): optical or RF
  \end{itemize}
  \item Target applications: disaster response, maritime navigation, defense surveillance $\Rightarrow$ \alert{sub-second deadlines}
\end{itemize}
\end{frame}

\begin{frame}{Problem and research question}
\framesubtitle{Task offloading and routing in LEO networks}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\small
\textbf{OEC constraints}
\begin{itemize}\setlength\itemsep{2pt}
  \item Limited battery / solar energy
  \item Constrained onboard hardware
  \item Dynamic topology, intermittent links
  \item Strict deadlines: \\ $T_{up}+T_{cpu}+T_{down}\le D_r$
\end{itemize}
\vspace{2pt}
\textbf{ILP trade-off}
\begin{itemize}\setlength\itemsep{2pt}
  \item Globally optimal, but NP-hard
  \item One ILP per task: too costly
  \item Batching: better solutions, but \alert{waiting time} eats deadlines
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\vspace{6pt}
\begin{colorblock}{maincolor}{Research question}
\small To what extent can a centralized optimization approach maintain global allocation efficiency in high-density satellite networks before queuing and solver overhead cause systemic performance degradation?
\end{colorblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Objective and contributions}
\framesubtitle{What this thesis delivers}
\begin{enumerate}\setlength\itemsep{4pt}
  \item \textbf{ILP model} for joint ISL routing and CPU allocation, switchable between \emph{Mapping Time} and \emph{Mapping Energy}
  \item \textbf{Dynamic batching} with Batch Size $\bs$ and Batch Timeout $\bt$, plus deadline compensation for queuing delay
  \item \textbf{Centralized orchestrator} in Python: SimPy (discrete-event), PuLP, HiGHS
  \item \textbf{Stress and bottleneck analysis}: arrival rates up to $\lam=10$ req/s
  \item \textbf{Benchmark} vs OrbitAware, DTS-base, DTS-Optimal and distributed ILP
\end{enumerate}
\end{frame}

\begin{frame}{Dynamic batching engine}
\framesubtitle{Event-driven, controlled by $\bs$ and $\bt$}
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm, every node/.style={font=\scriptsize}]

  \node[tbox, minimum width=2.3cm] (c) at (6.0,0) {Topology map\\multi-source\\Dijkstra};
  \node[tbox, minimum width=2.3cm] (d) at (8.7,0) {\textbf{ILP}\\PuLP + HiGHS};
  \node[tbox, minimum width=2.3cm] (e) at (11.4,0) {Dispatch\\Route / Retry /\\Reject};
  \foreach \s/\t in {a/b,b/c,c/d,d/e} \draw[-{Stealth}, thick, maincolor] (\s) -- (\t);
\end{tikzpicture}
\end{center}
\small
\begin{itemize}\setlength\itemsep{3pt}
  \item $\bs$ caps $|R|$, hence the $|R|\times|S|$ binary variables: bounded solve time
  \item $\bt$ caps the waiting time of any request: no starvation at low traffic
  \item Zero-delay barrier \texttt{yield Timeout(env,0)} commits all simultaneous events before reading telemetry (state consistency)
  \item Transient failures (no route, infeasible) are re-enqueued keeping the original $t_{arr}$
\end{itemize}
\end{frame}

\begin{frame}{Centralized ILP formulation}
\framesubtitle{Per batch $R$, over candidate satellites $S$}
\footnotesize
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Variables}\\
$x_{r,i}\in\{0,1\}$ assign $r$ to $i$; $y_r\in\{0,1\}$ reject $r$

\vspace{3pt}
\textbf{Objective} (Time-first)
\[
\min\; \alpha Z_{time}+\beta Z_{energy}+\gamma Z_{sunset}
\]
\vspace{-8pt}
\begin{align*}
Z_{time}   &= \sum_{r,i} x_{r,i}R_{r,i} + P_{drop}\sum_r y_r\\[2pt]
Z_{energy} &= \sum_{r,i} x_{r,i}E_{r,i} + P_{drop}\sum_r y_r
\end{align*}
\vspace{-6pt}
\begin{itemize}\setlength\itemsep{1.5pt}
  \item $R_{r,i}$: uplink latency + queues + CPU exec
  \item $E_{r,i}$: ISL forwarding ($E^{\text{up}}$) + CPU energy ($E^{\text{cpu}}$)
  \item $Z_{sunset} = \sum_{r,i} x_{r,i}\,(1 - \frac{t_{sunset,i}}{t_{max}})$
  \item Energy-first swaps $\alpha\leftrightarrow\beta$
\end{itemize}
\end{column}
\begin{column}{0.50\textwidth}
\textbf{Constraints}
\begin{align*}
&\textstyle\sum_{i} x_{r,i}+y_r=1 && \text{(uniqueness)}\\
&x_{r,i}R_{r,i}\le D_r && \text{(deadline)}\\
&\textstyle\sum_{r,i} x_{r,i}\,\epsilon_{r,i,k}\le B_k && \text{(battery)}\\
&\textstyle\sum_{r} x_{r,i}\le\lfloor |R|/|S|\rfloor+5 && \text{(load)}
\end{align*}
\vspace{2pt}
Rejected tasks get a \alert{cascade diagnosis}: no route, deadline, sunset, energy, capacity or global conflict.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Complexity and scalability}
\framesubtitle{Asymptotic analysis}
\small
\begin{itemize}\setlength\itemsep{4pt}
  \item \textbf{Topology mapping}: multi-source Dijkstra, $O(|AP|\cdot|S|\log|S|)$, since node degree $\le4$ gives $|E|=O(|S|)$
  \item \textbf{ILP size}: $|R|\cdot|S|+|R|$ binaries; with $\bs\le20$, $|S|\approx15$: at most 300 binaries and 330 rows
  \item \textbf{Solver time}: $\approx$0.30--0.60 ms, independent of the arrival rate $\lam$
  \item \textbf{Forwarding}: planned path $O(h_{up})$; greedy fallback $O(k\log k)\approx O(1)$ with $k\le4$
  \item \textbf{Telemetry}: $O(|S|)$ messages per update; $<0.1\%$ of ISL bandwidth for $|S|\le50$, but a limit for thousands of satellites
\end{itemize}
\end{frame}

\begin{frame}{Conclusions}
\framesubtitle{Answering the research question}
\small
\begin{colorblock}{maincolor}{The ceiling of centralized orchestration is architectural, not computational}
\whitebullets\footnotesize
\begin{itemize}\setlength\itemsep{2pt}
  \item With $\bs\le20$, HiGHS solves in $\approx$0.30--0.60 ms
  \item Onboard execution is bounded: $T_{exec}\approx0.17$--$0.48$ s
  \item Collapse above $\lam\ge6$ is driven by \textbf{ingress queuing}, which consumes the deadline slack
  \item Best centralized: small-batching ($\bs$=2, $\bt$=2 s) with $\alpha=1000$, $\gamma=10$: 54--67\% at $\lam=10$ (mixed)
\end{itemize}
\end{colorblock}
\begin{colorblock}[black]{sintefyellow}{Limitations}
\footnotesize
\begin{itemize}\setlength\itemsep{2pt}
  \item Batching dilemma for sub-second services
  \item Master node is a single point of failure (radiation, reboots, maneuvers)
  \item Telemetry overhead and stale state at LEO speeds
\end{itemize}
\end{colorblock}
\end{frame}
